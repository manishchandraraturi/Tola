   424→function ScribeConsole({
   425→  karigars,
   426→  onPrefill,
   427→}: {
   428→  karigars: StateResponse["karigars"];
   429→  onPrefill: (draft: ScribeDraft | null) => void;
   430→}) {
   431→  const [text, setText] = React.useState("25.4 grams 22K necklace set wastage 3, due 12 Nov, wedding order");
   432→  const [out, setOut] = React.useState<null | { heard: string; note: string; missing: string[]; parsed: ScribeDraft & { wastage?: number } }>(null);
   433→  const [busy, setBusy] = React.useState(false);
   434→  const { toast } = useToast();
   435→
   436→  async function run() {
   437→    setBusy(true);
   438→    setOut(null);
   439→    try {
   440→      const res = await fetch("/api/agents/scribe", {
   441→        method: "POST",
   442→        headers: { "Content-Type": "application/json" },
   443→        body: JSON.stringify({ text }),
   444→      });
   445→      const data = await res.json();
   446→      if (!res.ok) throw new Error(data.error);
   447→      setOut(data);
   448→    } catch (e) {
   449→      toast({ title: "Scribe could not listen", description: e instanceof Error ? e.message : "", variant: "destructive" });
   450→    } finally {
   451→      setBusy(false);
   452→    }
   453→  }
   454→
   455→  return (
   456→    <SketchBox tone="blue" variant="a" className="flex flex-col px-4 py-4">

   600→function SettlerConsole({
   601→  slips,
   602→  onRefresh,
   603→}: {
   604→  slips: SlipRow[];
   605→  onRefresh: () => void;
   606→}) {
   607→  const [slipId, setSlipId] = React.useState(slips[0]?.id ?? "");
   608→  const [out, setOut] = React.useState<null | { lines: string[]; upiLink: string; clockStartedAt: string; waitingFor: string[] }>(null);
   609→  const [busy, setBusy] = React.useState(false);
   610→  const { toast } = useToast();
   611→
   612→  React.useEffect(() => {
   613→    if (!slips.find((s) => s.id === slipId)) {
   614→      setSlipId(slips[0]?.id ?? "");
   615→      setOut(null);
   616→    }
   617→  }, [slips, slipId]);
   618→
   619→  return (
   620→    <SketchBox tone="salmon" variant="d" className="flex flex-col px-4 py-4">
   621→      <p className="font-sketch text-lg font-bold text-ink">Settler · statements, clock, UPI</p>
   622→      <p className="mt-1 font-hand text-sm text-ink-soft">
   623→        Builds both statements, starts the payment clock, waits for both approvals.
   624→      </p>
   625→      {slips.length === 0 ? (
   626→        <p className="mt-3 font-hand text-base text-ink-soft">
   627→          No returned slips waiting right now. Record a return on a slip first.
   628→        </p>
   629→      ) : (
   630→        <>
   631→          <select
   632→            value={slipId}
   633→            onChange={(e) => {
   634→              setSlipId(e.target.value);
   635→              setOut(null);
   636→            }}
   637→            className="mt-2 w-full border-[2.5px] border-ink bg-white px-3 py-2 font-hand text-base text-ink sketch-c outline-none"
   638→          >
   639→            {slips.map((s) => (
   640→              <option key={s.id} value={s.id}>
   641→                Slip #{s.number} · {s.job} · {s.karigarName}
   642→              </option>
   643→            ))}
   644→          </select>
   645→          <div className="mt-2">
   646→            <SketchButton
   647→              tone="salmon"
   648→              disabled={busy}
   649→              onClick={async () => {
   650→                setBusy(true);
   651→                try {
   652→                  const res = await fetch("/api/agents/settler", {
   653→                    method: "POST",
   654→                    headers: { "Content-Type": "application/json" },
   655→                    body: JSON.stringify({ slipId }),
   656→                  });
   657→                  const data = await res.json();
   658→                  if (!res.ok) throw new Error(data.error);
   659→                  setOut(data.statement);
   660→                  onRefresh();
   661→                } catch (e) {
   662→                  toast({ title: "Settler cannot start", description: e instanceof Error ? e.message : "", variant: "destructive" });
   663→                } finally {
   664→                  setBusy(false);
   665→                }
   666→              }}
   667→            >
   668→              {busy ? "Building…" : "Build statements + start the clock"}
   669→            </SketchButton>
   670→          </div>
   671→          {out && (
   672→            <div className="tola-pop mt-3 border-[2px] border-ink bg-white px-3 py-2 sketch-c">
   673→              <ul className="space-y-0.5 font-hand text-sm text-ink">
   674→                {out.lines.map((l) => (
   675→                  <li key={l}>• {l}</li>
   676→                ))}
   677→              </ul>
   678→              <p className="mt-2 font-hand text-xs text-ink-soft">
   679→                Clock started {fmtDateTime(out.clockStartedAt)} · waiting for{" "}
   680→                {out.waitingFor.length ? out.waitingFor.join(" + ") : "nobody"}
   681→              </p>
   682→            </div>
   683→          )}
   684→        </>
   685→      )}
   686→    </SketchBox>
   687→  );
   688→}
